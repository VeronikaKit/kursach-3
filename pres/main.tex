	(function() {
		'use strict';
		retryWhileError(function() {
			let s = sl(1, 200);       // расстояние
			let n = sl(1, 10);        // разница во времени
			let a = sl(1, 15);        // скорость течения
			let v = (a**2 * n**2 + 2 * s * a * n).sqrt();
			let x = v / n;
			genAssert(x > 0, 'Скорость не может быть отрицательной');
			genAssertZ1000(x, 'Скорость не должна быть слишком дробной');

		});

\begin{frame}{Попытки использования нейросетей для генерации тестов}
	\begin{itemize}
		\item Нейросети способны генерировать простейшие тесты без уточнений от программиста
		\item При усложнении задачи качество тестов существенно падает
		\item Нейросети не могут полностью заменить программиста в написании тестов
		\item Требуется экспертная оценка и доработка сгенерированных тестов человеком
	\end{itemize}
\end{frame}

\begin{frame}{Рефакторинг – зачем он нужен в JavaScript}
    \begin{itemize}
        \item \textbf{Отсутствие строгой типизации} → ошибки проявляются только во время выполнения.
        \item \textbf{Асинхронная природа} языка → «callback hell», нечитаемые цепочки.
        \item \textbf{Быстрое устаревание} кода (ECMAScript ежегодно обновляется).
        \item \textbf{Накопление технического долга} → замедление разработки, рост числа ошибок.
    \end{itemize}
    \vfill
    \begin{exampleblock}{Цель рефакторинга в проекте «Час ЕГЭ»}
        Улучшить структуру кода без изменения функциональности: повысить читаемость, убрать дублирование, облегчить добавление новых заданий.
    \end{exampleblock}
\end{frame}
